def pace_per_km(distance_km: float, time: str) -> str:
    parts = [int(p) for p in time.split(":")]
    if len(parts) == 2:
        hours, (minutes, seconds) = 0, parts
    elif len(parts) == 3:
        hours, minutes, seconds = parts
    else:
        raise ValueError("time must be mm:ss or hh:mm:ss")
    if distance_km <= 0:
        raise ValueError("distance must be positive")
    total = hours * 3600 + minutes * 60 + seconds
    pace = round(total / distance_km)
    return f"{pace // 60}:{pace % 60:02d}"
```
